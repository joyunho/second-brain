# 📔 일기 — 감정 · 회고

> YouKnow `#/daily` 화면. 일간 노트가 "무엇을 했나"라면, 일기는 "어땠나". 둘은 같은 날짜(`daily: [[YYYY-MM-DD]]`)로 연결된다.
> 일기 하나는 **(날짜, 종류)당 한 편**. 종류 · 기분 · 태그는 `50_일기/종류|기분|태그/` facet 노트로 선이 그어져 그물을 만든다.

```dataviewjs
const t = dv.date("today"), today = t.toFormat("yyyy-MM-dd");
const ds = v => !v ? null : (v.toFormat ? v.toFormat("yyyy-MM-dd") : String(v).slice(0, 10));
const ym = t.toFormat("yyyy-MM");
const all = dv.pages('"50_일기"').where(p => p.type === "일기" && p.date && !p.yk_deleted);   // 앱 휴지통 제외
const month = all.where(p => ds(p.date).startsWith(ym));
const days = new Set(month.map(p => ds(p.date)).array());
const dates = new Set(all.map(p => ds(p.date)).array());
// 앱 nU.streak: 오늘(없으면 어제)부터 거꾸로 연속으로 일기가 있는 날 수
let cur = dates.has(today) ? t : t.minus({ days: 1 }), streak = 0;
while (dates.has(cur.toFormat("yyyy-MM-dd"))) { streak++; cur = cur.minus({ days: 1 }); }
const photos = month.array().reduce((s, p) => s + Number(p.photos ?? 0), 0);
const todayDiary = all.where(p => ds(p.date) === today).first();
dv.paragraph(`**${t.month}월** · 일기 **${month.length}** · 기록 **${days.size}/${t.daysInMonth}일** · 사진 ${photos}장` + (streak >= 2 ? ` · 🔥 **${streak}일 연속**` : "") + (todayDiary ? ` · 오늘 일기 → ${todayDiary.file.link}` : " · 오늘 일기 아직 없음 — `Ctrl/Cmd+N` → 📔 일기"));
// 종류 칩 (50_일기/종류 노트 → 이번 달 편수)
const types = dv.pages('"50_일기/종류"').where(p => p.type === "종류" && !p.yk_deleted && !p.yk_missing).sort(p => p.file.name);
const typeChips = types.length
  ? types.map(ty => `[[${ty.file.name}]] ${month.where(p => (p.diary_type ?? "일반") === ty.file.name).length}`).array()
  : [...new Set(month.map(p => p.diary_type ?? "일반").array())].map(n => `${n} ${month.where(p => (p.diary_type ?? "일반") === n).length}`);
dv.paragraph(`종류: 전체 ${month.length} · ` + (typeChips.join(" · ") || "—"));
// 기분 칩 (50_일기/기분 노트 → 이번 달 편수, 점수 높은 순)
const moods = dv.pages('"50_일기/기분"').where(p => p.type === "기분" && !p.yk_deleted && !p.yk_missing).sort(p => -(p.score ?? 0));
const moodChips = moods.length
  ? moods.map(m => `${m.emoji ?? ""}[[${m.file.name}]] ${month.where(p => p.mood === m.file.name).length}`).array()
  : [...new Set(month.map(p => p.mood).array().filter(Boolean))].map(n => `${n} ${month.where(p => p.mood === n).length}`);
dv.paragraph(`기분: ` + (moodChips.join(" · ") || "—"));
// 운동 종류 요약 (앱 am: N월 운동 N회 · 총 N분 · 최다 부위)
const workouts = month.where(p => p.duration_min || p.parts);
if (workouts.length) {
  const mins = workouts.array().reduce((s, p) => s + Number(p.duration_min ?? 0), 0);
  const cnt = {}; for (const p of workouts) for (const part of (Array.isArray(p.parts) ? p.parts : [p.parts].filter(Boolean))) cnt[part] = (cnt[part] ?? 0) + 1;
  const top = Object.entries(cnt).sort((a, b) => b[1] - a[1])[0];
  dv.paragraph(`🏋️ ${t.month}월 운동 **${workouts.length}회** · 총 **${mins}분**` + (top ? ` · 최다 부위 **${top[0]}** ${top[1]}회` : ""));
}
```

## 이번 달 달력
```dataviewjs
const t = dv.date("today"), today = t.toFormat("yyyy-MM-dd");
const ds = v => !v ? null : (v.toFormat ? v.toFormat("yyyy-MM-dd") : String(v).slice(0, 10));
const ym = t.toFormat("yyyy-MM");
const EMO = { 5: "🤩", 4: "🙂", 3: "😐", 2: "😕", 1: "😞" };
const moodEmoji = {}; for (const m of dv.pages('"50_일기/기분"').where(p => p.type === "기분")) moodEmoji[m.file.name] = m.emoji ?? EMO[m.score] ?? "";
const byDay = {};
for (const p of dv.pages('"50_일기"').where(p => p.type === "일기" && p.date && !p.yk_deleted && ds(p.date).startsWith(ym))) (byDay[ds(p.date)] ??= []).push(p);
const first = dv.date(`${ym}-01`), lead = first.weekday % 7;   // 앱 달력과 같은 일요일 시작
const cells = []; for (let i = 0; i < lead; i++) cells.push("");
for (let d = 1; d <= t.daysInMonth; d++) {
  const key = `${ym}-${String(d).padStart(2, "0")}`, ps = byDay[key] ?? [];
  const mark = ps.map(p => `${moodEmoji[p.mood] ?? (p.mood ? "·" : "")}${p.file.link.withDisplay((p.diary_type ?? "일반") === "일반" ? "일기" : p.diary_type).markdown()}`).join("<br>");
  const photos = ps.reduce((s, p) => s + Number(p.photos ?? 0), 0);
  cells.push(`${key === today ? "**" + d + "**" : d}${ps.length ? "<br>" + mark : ""}${photos ? ` 📷${photos}` : ""}`);
}
while (cells.length % 7) cells.push("");
const rows = []; for (let i = 0; i < cells.length; i += 7) rows.push(cells.slice(i, i + 7));
dv.table(["일", "월", "화", "수", "목", "금", "토"], rows);
```

## 📷 갤러리 (최근 대표 사진)
```dataviewjs
const ds = v => !v ? null : (v.toFormat ? v.toFormat("yyyy-MM-dd") : String(v).slice(0, 10));
const linkPath = v => !v ? null : (v.path ? String(v.path) : String(v).replace(/^!?\[\[|\]\]$/g, "").split("|")[0]);
const withCover = dv.pages('"50_일기"').where(p => p.type === "일기" && p.cover && p.date && !p.yk_deleted).sort(p => ds(p.date), "desc").slice(0, 12).array();
if (!withCover.length) dv.paragraph("_아직 사진이 있는 일기가 없어요. 앱에서 올린 사진은 가져오기가 `_Attachments/YouKnow/daily/`에 받아 두고 일기의 `cover`로 연결해요._");
else dv.table(["사진", "일기", "날짜"], withCover.map(p => [`![[${linkPath(p.cover).split("/").pop()}|160]]`, p.file.link, ds(p.date)]));
```

## 이번 달 목록 (일요일 시작 주 묶음)
```dataviewjs
const t = dv.date("today");
const ds = v => !v ? null : (v.toFormat ? v.toFormat("yyyy-MM-dd") : String(v).slice(0, 10));
const dows = ["일","월","화","수","목","금","토"];
const ym = t.toFormat("yyyy-MM");
const month = dv.pages('"50_일기"').where(p => p.type === "일기" && p.date && !p.yk_deleted && ds(p.date).startsWith(ym)).sort(p => ds(p.date), "desc").array();
if (!month.length) dv.paragraph("이 달엔 일기가 없어요. 오늘 일기 쓰기로 시작해보세요.");
const weeks = {};
for (const p of month) { const d = dv.date(ds(p.date)); const sun = d.minus({ days: d.weekday % 7 }); const k = sun.toFormat("yyyy-MM-dd"); (weeks[k] ??= []).push(p); }
for (const k of Object.keys(weeks).sort().reverse()) {
  const sun = dv.date(k), sat = sun.plus({ days: 6 });
  dv.header(3, `${sun.toFormat("M/d")} ~ ${sat.toFormat("M/d")}`);
  dv.table(["날", "일기", "종류", "기분", "태그", "📷"], weeks[k].map(p => {
    const d = dv.date(ds(p.date));
    const sub = p.duration_min || p.parts
      ? [Array.isArray(p.parts) ? p.parts.join("·") : p.parts, p.duration_min ? `${p.duration_min}분` : null, p.intensity].filter(Boolean).join(" · ")
      : (Array.isArray(p.tags) ? p.tags.join(" · ") : (p.tags ?? ""));
    return [`${d.day} (${dows[d.weekday % 7]})`, p.file.link, p.diary_type ?? "일반", p.mood ?? "", sub, p.photos ?? ""];
  }));
}
```

## 최근 일기
```dataview
TABLE WITHOUT ID file.link AS "일기", date AS "날짜", diary_type AS "종류", mood AS "기분", mood_score AS "점수", tags AS "태그", photos AS "📷"
FROM "50_일기"
WHERE type = "일기" AND !yk_deleted
SORT date DESC
LIMIT 30
```

## 기분 추이
```dataviewjs
const ds = v => !v ? null : (v.toFormat ? v.toFormat("yyyy-MM-dd") : String(v).slice(0, 10));
const t = dv.date("today");
const all = dv.pages('"50_일기"').where(p => p.type === "일기" && p.date && !p.yk_deleted);
const recent = all.where(p => p.mood_score && dv.date(ds(p.date)) >= t.minus({ days: 30 }));
const avg = recent.length ? (recent.array().reduce((s, p) => s + Number(p.mood_score), 0) / recent.length).toFixed(1) : "–";
const EMO = { 5: "🤩", 4: "🙂", 3: "😐", 2: "😕", 1: "😞" };
dv.paragraph(`최근 30일 ${recent.length}편 · 평균 기분 **${avg}** / 5 ${avg !== "–" ? EMO[Math.round(avg)] ?? "" : ""}  _(최고 5 · 좋음 4 · 보통 3 · 별로 2 · 힘듦 1)_`);
// 월별 카운트 (최근 12개월)
const rows = [];
for (let i = 0; i < 12; i++) {
  const m = t.minus({ months: i }); const ym = m.toFormat("yyyy-MM");
  const inM = all.where(p => ds(p.date).startsWith(ym));
  if (!inM.length && i > 0) continue;
  const scored = inM.where(p => p.mood_score);
  const a = scored.length ? (scored.array().reduce((s, p) => s + Number(p.mood_score), 0) / scored.length).toFixed(1) : "–";
  const moodCount = {}; for (const p of inM) if (p.mood) moodCount[p.mood] = (moodCount[p.mood] ?? 0) + 1;
  const topMood = Object.entries(moodCount).sort((x, y) => y[1] - x[1])[0];
  rows.push([`${m.year}년 ${m.month}월`, inM.length, new Set(inM.map(p => ds(p.date)).array()).size + `/${m.daysInMonth}일`, a, topMood ? `${topMood[0]} ${topMood[1]}` : "—", inM.array().reduce((s, p) => s + Number(p.photos ?? 0), 0)]);
}
dv.table(["월", "일기", "기록", "평균 기분", "최다 기분", "📷"], rows);
```

## 종류 · 기분 · 태그
```dataview
TABLE WITHOUT ID file.link AS "값", type AS "facet", length(file.inlinks) AS "일기 수"
FROM "50_일기/종류" OR "50_일기/기분" OR "50_일기/태그"
WHERE (type = "종류" OR type = "기분" OR type = "태그") AND !yk_deleted AND !yk_missing
SORT type ASC, length(file.inlinks) DESC
```

## 쓰는 법
- **새 일기**: `Ctrl/Cmd+N` → 📔 일기 → 종류(기본 일반) → 날짜(기본 오늘, "어제" 허용) → 기분 → 태그. 파일은 `50_일기/YYYY/YYYY-MM-DD 일기.md`(일반) 또는 `YYYY-MM-DD 운동.md`. 같은 (날짜, 종류)가 이미 있으면 그 노트를 연다.
- 속성은 평문 + 링크 쌍이다: `mood: 좋음` + `기분: "[[좋음]]"`, `tags: [운동, 친구]` + `태그: ["[[운동]]", "[[친구]]"]`, `diary_type: 운동` + `종류: "[[운동]]"`. 평문은 Dataview용, 링크는 그래프 선용.
- **폰**: YouKnow 앱에서 쓰고 사진을 올리면 밤에 `50_일기/`와 `_Attachments/YouKnow/daily/`로 들어온다. 본문은 `%% yk:body %%` 블록 안에만 다시 쓰이고, `## 생각 · 연결` 아래는 내 것이다.
- **옵시디언 → 앱**: `import_youknow.py --export-diaries-for-youknow <폴더>`가 앱 ".md 일기 가져오기" 형식으로 써 준다(→ [[🔌 연동]]).
